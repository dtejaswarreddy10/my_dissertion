      NOTE: do NOT use synth/asic/asic_power_sweep.sh as-is. It calls
      synth_pe_mixed.tcl (aborts on check -assert) and builds SAIF from a finished
      .wlf (produces nothing). Use the steps below, or rewrite the script.

[ ] A1. Regenerate pe_mixed netlists (only if needed)
      cd synth/asic && yosys -c synth_pe_mixed_abcfix.tcl && cd ../..
      -> reports/pe_mixed_{nogate,gated}_abcfix.v and *_abcfix_stat.txt

[ ] A2. SAIF per sparsity point (0/25/50/75%), both netlists, live POWER=1 run
      cd tb
      for s in 0.00 0.25 0.50 0.75; do
        tag=sp$(printf '%02d' $(echo "$s*100/1" | bc))
        for cfg in nogate gated; do
          make clean
          SAIF_OUT=$NETD/activity_${cfg}_${tag}.saif \
          make SIM=questa TEST=pe_mixed_gl POWER=1 \
               GL_NETLIST=$NETD/pe_mixed_${cfg}_abcfix.v GL_LIB_V=$GL_VERILOG \
               TESTCASE=test_power_stimulus COCOTB_TESTCASE=test_power_stimulus \
               TB_SPARSITY=$s TB_POWER_CYCLES=4000 \
            | tee logs/gl_power_${cfg}_${tag}.log
          [ -s $NETD/activity_${cfg}_${tag}.saif ] || echo "!! EMPTY SAIF: $cfg $tag"
        done
      done
      cd ..
      Check: each log ran only test_power_stimulus; "measured zero-operand" ~ target.
      Optional: repeat with TB_POWER_MODE=int4x2 for an INT4x2 curve.

[ ] A3. Power per SAIF
      OpenSTA (timing + power):
        cd synth/asic
        for s in 00 25 50 75; do for cfg in nogate gated; do
          NETLIST=$NETD/pe_mixed_${cfg}_abcfix.v SAIF=$NETD/activity_${cfg}_sp${s}.saif \
          TOP=pe_mixed CLK_NS=4.0 sta -exit -f sta_pe.tcl | tee logs/sta_${cfg}_sp${s}.log
        done; done
        (If read_saif warns on scope, set SAIF_SCOPE=pe_mixed or tb/pe_mixed.)
      gl_power.py fallback (ratio only, load-cap model):
        for s in 00 25 50 75; do for cfg in nogate gated; do
          ./gl_power.py --liberty $FREEPDK45_LIB --netlist $NETD/pe_mixed_${cfg}_abcfix.v \
             --saif $NETD/activity_${cfg}_sp${s}.saif --json $NETD/power_${cfg}_sp${s}.json
        done
        ./gl_power.py --compare $NETD/power_nogate_sp${s}.json $NETD/power_gated_sp${s}.json
        done
